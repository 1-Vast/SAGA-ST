import scanpy as sc
import pandas as pd

data_dir = "/root/autodl-tmp/cancer/SEDR_analyses/data/BRCA1/V1_Human_Breast_Cancer_Block_A_Section_1"
meta_path = "/root/autodl-tmp/cancer/SEDR_analyses/data/BRCA1/metadata.tsv"
out_path = "/root/autodl-tmp/cancer/breast_with_gt.h5ad"

# 读取 Visium
adata = sc.read_visium(
    data_dir,
    count_file="filtered_feature_bc_matrix.h5"
)
adata.var_names_make_unique()

# 读取标签
meta = pd.read_csv(meta_path, sep="\t").set_index("ID")

# 合并两级标签
adata.obs["annot_type"] = meta.reindex(adata.obs_names)["annot_type"]
adata.obs["fine_annot_type"] = meta.reindex(adata.obs_names)["fine_annot_type"]

# 只保留有细粒度标签的 spot
adata = adata[adata.obs["fine_annot_type"].notna()].copy()

print("最终对象：", adata)
print("\nobs 列：", list(adata.obs.columns))
print("\n粗粒度标签分布：")
print(adata.obs["annot_type"].value_counts())
print("\n细粒度标签分布：")
print(adata.obs["fine_annot_type"].value_counts())

adata.write_h5ad(out_path)
print("\n已保存到：", out_path)
